# Sorting Resort Hotel bookings 

resort_hb_q3 <- hotel_booking %>%
  filter(hotel == "Resort Hotel") %>%
  mutate(
  lead_time_groups = case_when(
    lead_time <= 14 ~ "Last Minute",
    lead_time <= 90 ~ "Short Term",
    lead_time <= 200 ~ "Medium Term",
    TRUE ~ "Advance Term"
  ),
  booking_channel = case_when(
    market_segment == "Online TA" | market_segment == "Offline TA" ~ "External",
    market_segment == "Direct" ~ "Direct",
    market_segment == "Corporate" ~ "Corporate",
    TRUE ~ "Other"
  )
  )

resort_risk_table <- resort_hb_q3 %>%
  group_by(booking_channel, lead_time_groups) %>%
  summarise(
    total_bookings = n(),
    total_canceled = sum(is_canceled),
    cancellation_rate = (total_canceled / total_bookings) * 100,
    mean_adr = mean(adr)
    
  ) %>%
  ungroup()

# Modelling for Resort Hotel 

library(tidyverse)

library(tidymodels)

resort_model <- resort_hb_q3 %>%
  mutate(
    is_canceled = factor(is_canceled, levels = c("1", "0")),
    booking_channel = as.factor(booking_channel)
  ) %>%
  select(is_canceled, lead_time_groups, booking_channel) %>%
  drop_na()

set.seed(123)
resort_split <- initial_split(resort_model, prop = 0.80)
resort_train <- training(resort_split)
resort_test <- testing(resort_split)

resort_recipe <- recipe(is_canceled ~ lead_time_groups + booking_channel, data = resort_train) %>%
  step_dummy(all_nominal_predictors())

resort_mod <- logistic_reg() %>%
  set_engine("glm")

resort_wflow <- workflow() %>%
  add_recipe(resort_recipe) %>%
  add_model(resort_mod)

resort_fit <- resort_wflow %>%
  fit(data = resort_train)

tidy(resort_fit)
